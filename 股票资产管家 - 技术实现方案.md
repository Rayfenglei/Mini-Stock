# 股票资产管家 - 技术实现方案

## 1. 项目概述

### 1.1 技术选型
| 技术类别 | 技术栈 | 说明 |
|---------|--------|------|
| 前端框架 | 微信小程序原生 | 使用原生 WXML/WXSS/JavaScript |
| UI 组件库 | TDesign | 腾讯开源组件库，风格统一 |
| 图表库 | F2 | 蚂蚁金服移动端图表库 |
| 后端服务 | 微信云开发 | 云函数 + 云数据库 + 云存储 |
| 股票数据 | 腾讯财经 API | 免费、稳定、实时 |
| 基金数据 | 天天基金 API | 基金净值数据，T+1 更新 |
| 黄金数据 | 金投网 API | 现货黄金价格 |

### 1.1.1 数据源说明

| 资产类型 | 数据源 | API 示例 | 更新频率 |
|---------|--------|---------|---------|
| 股票 | 腾讯财经 | `https://qt.gtimg.cn/q=sh600000` | 实时 |
| 基金 | 天天基金 | `https://fundgz.1234567.com.cn/js/{fundCode}.js` | T+1 |
| 黄金 | 金投网 | `https://api.jintou.com/gold/price` | 5 分钟 |

#### 页面逻辑（assetDetail.js）
```javascript
const app = getApp();
const api = require('../../utils/api');
const format = require('../../utils/format');

Page({
  data: {
    currentAccount: {},
    todayProfit: 0,
    todayRate: 0,
    totalProfit: 0,
    totalProfitRate: 0,
    totalAmount: 0,
    costAmount: 0,
    holdings: [],
    sortAsc: false,
    sortBy: 'marketValue'
  },

  onLoad(options) {
    const accountId = options.id || '';
    if (accountId) {
      this.loadAccountData(accountId);
    } else {
      this.loadDefaultAccount();
    }
  },

  onPullDownRefresh() {
    this.refreshData();
  },

  onShow() {
    // 页面显示时刷新数据
    this.refreshData();
  },

  // 加载默认账户
  async loadDefaultAccount() {
    try {
      const accounts = await api.getAccounts();
      if (accounts.length > 0) {
        this.loadAccountData(accounts[0]._id);
      } else {
        wx.redirectTo({ url: '/pages/index/index' });
      }
    } catch (error) {
      console.error('加载账户失败', error);
      wx.showToast({ title: '加载失败', icon: 'none' });
    }
  },

  // 加载账户数据
  async loadAccountData(accountId) {
    wx.showLoading({ title: '加载中...' });
    
    try {
      const [account, holdings] = await Promise.all([
        api.getAccountById(accountId),
        api.getHoldings(accountId)
      ]);

      this.setData({ currentAccount: account });
      this.calculateTotalData(holdings);
      this.setData({ holdings });
      
      // 缓存当前账户 ID
      wx.setStorageSync('currentAccountId', accountId);
    } catch (error) {
      console.error('加载数据失败', error);
      wx.showToast({ title: '加载失败', icon: 'none' });
    } finally {
      wx.hideLoading();
      wx.stopPullDownRefresh();
    }
  },

  // 计算总数据
  calculateTotalData(holdings) {
    let todayProfit = 0;
    let totalProfit = 0;
    let totalAmount = 0;
    let costAmount = 0;

    holdings.forEach(item => {
      todayProfit += item.todayProfit || 0;
      totalProfit += item.profit || 0;
      totalAmount += item.marketValue || 0;
      costAmount += item.costAmount || 0;
    });

    const totalProfitRate = costAmount > 0 
      ? (totalProfit / costAmount) * 100 
      : 0;
    
    const todayRate = totalAmount > 0 
      ? (todayProfit / totalAmount) * 100 
      : 0;

    this.setData({
      todayProfit,
      todayRate,
      totalProfit,
      totalProfitRate,
      totalAmount,
      costAmount
    });
  },

  // 刷新数据
  async refreshData() {
    const accountId = this.data.currentAccount._id;
    if (accountId) {
      await this.loadAccountData(accountId);
    }
  },

  // 切换排序
  toggleSort() {
    const { holdings, sortAsc } = this.data;
    const newSortAsc = !sortAsc;
    
    const sorted = [...holdings].sort((a, b) => {
      return newSortAsc 
        ? a.marketValue - b.marketValue 
        : b.marketValue - a.marketValue;
    });

    this.setData({
      holdings: sorted,
      sortAsc: newSortAsc
    });
  },

  // 显示账户选择器
  async showAccountPicker() {
    try {
      const accounts = await api.getAccounts();
      const accountNames = accounts.map(a => 
        `${a.name} | ${a.broker}${a.accountCode}`
      );

      wx.showActionSheet({
        itemList: accountNames,
        success: (res) => {
          this.loadAccountData(accounts[res.tapIndex]._id);
        }
      });
    } catch (error) {
      wx.showToast({ title: '获取账户失败', icon: 'none' });
    }
  },

  // 编辑账户
  editAccount() {
    const { currentAccount } = this.data;
    wx.navigateTo({
      url: `/pages/editAccount/editAccount?id=${currentAccount._id}`
    });
  },

  // 点击持仓卡片
  onHoldingTap(e) {
    const { _id } = e.currentTarget.dataset;
    wx.navigateTo({
      url: `/pages/holdingDetail/holdingDetail?id=${_id}`
    });
  },

  // 添加股票
  goToAddStock() {
    const { currentAccount } = this.data;
    wx.navigateTo({
      url: `/pages/addStock/addStock?accountId=${currentAccount._id}`
    });
  }
});
```

      // 计算盈亏百分比（用于进度条）
      const { profit, costAmount } = newData;
      const maxLoss = costAmount; // 最大亏损为成本价
      const percent = Math.abs(profit) / maxLoss * 100;
      
      this.setData({
        profitPercent: Math.min(percent, 100)
      });

      // 格式化更新时间
      const date = new Date(newData.updateTime);
      const month = date.getMonth() + 1;
      const day = date.getDate();
      this.setData({
        updateTime: `${month}月${day}日`
      });
    }
  }
});

// 刷新股票行情
async function refreshStockQuotes(holdings) {
  const stockCodes = holdings.map(h => {
    // 根据股票代码判断交易所
    const prefix = h.stockCode.startsWith('6') ? 'sh' : 'sz';
    return prefix + h.stockCode;
  });

  // 调用腾讯财经 API
  const response = await fetchStockQuotes(stockCodes);

  return holdings.map(holding => {
    const prefix = holding.stockCode.startsWith('6') ? 'sh' : 'sz';
    const quote = response[prefix + holding.stockCode];

    if (!quote) return holding;

    const currentPrice = quote.currentPrice;
    const preClose = quote.preClose; // 昨收
    const todayChange = currentPrice - preClose;
    const todayChangePercent = (todayChange / preClose) * 100;

    const marketValue = currentPrice * holding.shares;
    const todayProfit = todayChange * holding.shares;
    const todayRate = todayChangePercent;
    const profit = (currentPrice - holding.costPrice) * holding.shares;
    const profitRate = (profit / holding.costAmount) * 100;

    return {
      ...holding,
      currentPrice,
      marketValue,
      todayProfit,
      todayRate,
      profit,
      profitRate
    };
  });
}

// 调用股票 API（腾讯财经）
async function fetchStockQuotes(codes) {

  const result = {};

/**
 * 刷新基金净值数据
 * 基金净值 T+1 更新，每个交易日晚上更新
 */
async function refreshFundQuotes(holdings) {
  const fundCodes = holdings
    .filter(h => h.assetType === 'fund')
    .map(h => h.assetCode);

  const response = await fetchFundNetValues(fundCodes);

  return holdings.map(holding => {
    if (holding.assetType !== 'fund') return holding;

    const fundData = response[holding.assetCode];
    if (!fundData) return holding;

    // 基金使用净值计算收益
    const currentPrice = fundData.netValue; // 单位净值
    const costAmount = holding.costAmount;
    const shares = holding.shares;
    const marketValue = currentPrice * shares;
    
    // 估算净值（盘中估值）
    const estimate = fundData.estimate || currentPrice;
    
    // 计算收益
    const profit = (currentPrice - holding.costPrice) * shares;
    const profitRate = (profit / costAmount) * 100;
    
    // 今日涨跌幅（对比上一交易日净值）
    const preNetValue = fundData.preNetValue || holding.costPrice;
    const todayRate = ((currentPrice - preNetValue) / preNetValue) * 100;
    const todayProfit = todayRate * marketValue / 100;

    return {
      ...holding,
      currentPrice,
      marketValue,
      todayProfit,
      todayRate,
      profit,
      profitRate,
      fundNetValue: currentPrice,
      fundEstimate: estimate,
      fundUpdateTime: fundData.updateTime || new Date()
    };
  });
}

// 获取基金净值

          method: 'GET',
          headers: {

          }
        }

      const data = JSON.parse(jsonStr);

      result[code] = {
        netValue: parseFloat(data.dwjz), // 单位净值
        preNetValue: parseFloat(data.zrqz), // 上一交易日净值
        estimate: parseFloat(data.gsz), // 估算值

      .then(res => res.data[0]);

          costAmount: totalCost,

        todayProfit: 0,
        todayRate: 0,
        profit: 0,
        profitRate: 0,
        createTime: now,
        updateTime: now

        holdingData.goldType = goldType || 'au9999';

      // 基金特有字段
